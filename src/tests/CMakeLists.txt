# ── The container C ABI ─────────────────────────────────────────────────────
#
# test_iwi_c_api.cpp exercises the exported C surface at runtime - sizing,
# parsing, the component accessor, decode, metadata, and every rejection path -
# and returns non-zero on any mismatch, so ctest fails on regression.
#
# test_abi_constants.cpp pins every on-disk offset and every enumeration
# literal at compile time, which is what stops the C and C++ halves of
# ImagePluginDef.h from drifting apart.
#
# Both link the same static library an external consumer would: one header, one
# library, C linkage, no module.
if(TARGET iw_iwi_file)
    add_executable(iwi_c_api_test abi/test_iwi_c_api.cpp)

    set_target_properties(iwi_c_api_test PROPERTIES
        CXX_STANDARD 17
        CXX_STANDARD_REQUIRED ON
    )

    target_link_libraries(iwi_c_api_test PRIVATE iw_iwi_file)

    add_test(NAME iwi_c_api_test COMMAND iwi_c_api_test)

    add_executable(iwi_abi_constants_test abi/test_abi_constants.cpp)

    set_target_properties(iwi_abi_constants_test PROPERTIES
        CXX_STANDARD 17
        CXX_STANDARD_REQUIRED ON
    )

    target_link_libraries(iwi_abi_constants_test PRIVATE iw_iwi_file)

    add_test(NAME iwi_abi_constants_test COMMAND iwi_abi_constants_test)
endif()

# ── C11 compatibility gate ──────────────────────────────────────────────────
#
# ImagePluginDef.h and IwiFile.h claim to be C headers.  That claim is only
# worth something if a C compiler accepts them, which is how Tension binds the
# format through FFI and how a C engine would consume it.  Without this gate a
# declaration can quietly become C++-only and nobody notices until a downstream
# consumer's build breaks - which is exactly what happened when five
# `enum X : uint8_t` declarations were introduced: they are C23, and the
# project's own C consumers are C11.
#
# The translation unit is generated into the *build* tree, not the repository,
# so the "no .c source in the repo" rule holds while the safety net remains.
# It also _Static_asserts the C half of every duplicated enumeration value
# against its literal, which is the other side of the check that
# test_abi_constants.cpp performs on the C++ half.
#
# Runs as a ctest test (so every job runs it) and as a build target (so a
# developer can run it without going through ctest).
set(_iwi_gate_src "${CMAKE_CURRENT_BINARY_DIR}/iwi_c_header_gate.c")
# The gate compiles IwiFile.h (src/format, the format authority) and
# ImagePluginDef.h (src/imageviewer, the plugin ABI it includes).
set(_iwi_gate_inc "${CMAKE_SOURCE_DIR}/src/format")
set(_iwi_gate_plugin_inc "${CMAKE_SOURCE_DIR}/src/imageviewer")

# The body lives in a header - abi/iwi_c_gate_body.h - so the CI script can
# compile exactly the same assertions without configuring the project.  This
# generates only a one-line stub into the build tree, which keeps the repository
# free of .c files.
file(WRITE "${_iwi_gate_src}"
"/* Generated by src/tests/CMakeLists.txt - do not edit.
 *
 * A translation unit whose only job is to be compiled as C11.  The assertions
 * are in iwi_c_gate_body.h so the standalone CI gate compiles the same text.
 */
#include \"iwi_c_gate_body.h\"
")

set(_iwi_gate_body_dir "${CMAKE_CURRENT_SOURCE_DIR}/abi")
if(MSVC)
    #
    # /TC forces C for this file.  The implicit include directories are passed
    # explicitly because ctest does not necessarily run with the Visual Studio
    # developer environment, and cl finds <stddef.h> only through the INCLUDE
    # variable - a gate that fails because of a missing environment is a gate
    # people learn to ignore.
    #
    set(_iwi_gate_sys_inc "")
    foreach(_dir IN LISTS CMAKE_CXX_IMPLICIT_INCLUDE_DIRECTORIES)
        list(APPEND _iwi_gate_sys_inc /I${_dir})
    endforeach()

    # The compiler needs its own INCLUDE (the CRT and Windows SDK headers), and
    # an environment is not something ctest guarantees: running this test from a
    # plain shell failed with "Cannot open include file: 'stddef.h'" while the
    # same command worked from a developer prompt.  A gate that passes or fails
    # depending on which shell invoked ctest is worse than no gate.
    #
    # So the build system runs the compile.  `cmake --build --target
    # iwi_c_header_gate` gets whatever environment the generator uses - a
    # developer prompt for Ninja, MSBuild for the Visual Studio generator - which
    # is the environment the project itself is compiled in.
    set(_iwi_gate_cmd
        "${CMAKE_CXX_COMPILER}" /nologo /TC /std:c11 /W4
        ${_iwi_gate_sys_inc}
        /I${_iwi_gate_inc}
        /I${_iwi_gate_plugin_inc}
        /I${_iwi_gate_body_dir}
        /c ${_iwi_gate_src}
        /Fo${CMAKE_CURRENT_BINARY_DIR}/iwi_c_header_gate.obj)

    add_test(NAME iwi_c_header_gate
             COMMAND ${CMAKE_COMMAND} --build "${CMAKE_BINARY_DIR}"
                     --target iwi_c_header_gate)
    add_custom_target(iwi_c_header_gate
        COMMAND ${_iwi_gate_cmd}
        COMMENT "Compiling the IWI/plugin headers as C11"
        VERBATIM)
else()
    # -pedantic-errors turns "this is a C23 extension" into a hard error, which
    # is the whole point: the consumers are C11.
    set(_iwi_gate_cmd
        "${CMAKE_CXX_COMPILER}" -x c -std=c11 -pedantic-errors
        -Wall -Wextra -Werror
        -I ${_iwi_gate_inc}
        -I ${_iwi_gate_plugin_inc}
        -I ${_iwi_gate_body_dir}
        -fsyntax-only ${_iwi_gate_src})

    add_test(NAME iwi_c_header_gate COMMAND ${_iwi_gate_cmd})
    add_custom_target(iwi_c_header_gate
        COMMAND ${_iwi_gate_cmd}
        COMMENT "Compiling the IWI/plugin headers as C11"
        VERBATIM)
endif()

# ── Windows shell thumbnail provider ────────────────────────────────────────
#
# Covers the parts of IwiThumbnailProvider that can be checked without a
# desktop: which files are accepted, how they are decimated, what the resulting
# bytes mean, and whether the CLSID the DLL was built with still matches the
# one the install scripts write (a mismatch fails silently in Explorer).
#
# C++23 because the subject is a module: the test imports IwiThumbnail and
# IwiFormat exactly as the provider does.  Gated on the library target, which
# exists only on Windows with IMAGEVIEWER_BUILD_THUMBNAIL_PROVIDER=ON.
if(TARGET iw_iwi_thumbnail)
    find_package(doctest REQUIRED)

    add_executable(iview_shell_tests
        test_iwi_thumbnail.cpp
    )

    set_property(TARGET iview_shell_tests PROPERTY CXX_STANDARD 23)

    target_include_directories(iview_shell_tests PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${CMAKE_CURRENT_SOURCE_DIR}/../imageviewer
    )

    target_link_libraries(iview_shell_tests PRIVATE
        iw_iwi_thumbnail
        doctest::doctest
    )

    # repo_root.hpp walks up from the executable to find the checkout; no path
    # needs to be baked in, so the test works from any generator's output
    # directory.
    add_test(NAME iview_shell_tests COMMAND iview_shell_tests)

    # The COM server's lifetime contract needs the real DLL: DllGetClassObject,
    # DllCanUnloadNow and LockServer are only meaningful across a module
    # boundary.  That source loads it by the path below.
    if(TARGET IwiThumbnailProvider)
        target_sources(iview_shell_tests PRIVATE
            test_thumbnail_provider_com.cpp
        )
        target_compile_definitions(iview_shell_tests PRIVATE
            IW_THUMBNAIL_PROVIDER_DLL="$<TARGET_FILE:IwiThumbnailProvider>"
        )
        # The genex above does not create an ordering edge by itself.
        add_dependencies(iview_shell_tests IwiThumbnailProvider)
    endif()
endif()
